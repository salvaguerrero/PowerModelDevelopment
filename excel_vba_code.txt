Function populateNode(nodeId As Long, objName As String) As String
    Dim idList As String
    Dim nodeCol As Long

    Set ws = ThisWorkbook.Sheets(objName)
    idList = "["
    
    'identifiying node_id column
    For nodeCol = 1 To ws.Columns.Count
        If LCase(ws.Cells(1, nodeCol).value) = "node_id" Then
            Exit For
        ElseIf ws.Cells(1, nodeCol).value = "" Then
            nodeCol = 0
            Exit For
        End If
    Next nodeCol
    For i = 2 To ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
        If ws.Cells(i, 1).value = "" Then
            Exit For
        End If
        
        If ws.Cells(i, nodeCol).value = nodeId Then
            idList = idList & CStr(ws.Cells(i, 1).value) & ","   '  save generators id
        End If
    Next i
    
    If Right(idList, 1) = "," Then idList = Left(idList, Len(idList) - 1)
    idList = idList & "]"
    
    If idList = "[]" Then idList = 0

    populateNode = idList
End Function

Function ConvertToJSON(dataDict As Collection, objName As String) As String
    Dim json As String
    Dim key As Variant
    Dim item As Variant
    Dim tempDict As Collection
    Dim heaDict As Collection
    Dim i As Long

    Set heaDict = dataDict(1)
    objName = LCase(objName)
    json = """" & objName & """:{"
    For i = 1 To dataDict(2).Count ' for each object
        
        Set tempDict = dataDict(2).item(i)
        If objName <> "Execution" Then
            json = json & """" & tempDict.item(1) & """:{"
            For j = 2 To heaDict.Count       ' for each boject property
                json = json & """" & heaDict.item(j) & """: """ & tempDict.item(j) & """" & ","
            Next j
            If Right(json, 1) = "," Then json = Left(json, Len(json) - 1)
            json = json & "},"
        Else
            For j = 1 To heaDict.Count       ' for each boject property
                json = json & """" & heaDict.item(j) & """: """ & tempDict.item(j) & """" & ","
            Next j
            If Right(json, 1) = "," Then json = Left(json, Len(json) - 1)
            
        End If

    Next i

    ' Remove last comma and close the JSON string
    If Right(json, 1) = "," Then json = Left(json, Len(json) - 1)
    json = json & "},"

    ConvertToJSON = json
End Function
